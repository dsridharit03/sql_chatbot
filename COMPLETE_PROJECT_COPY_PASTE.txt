import pyodbc
from django.conf import settings

class DatabaseService:
    
    @staticmethod
    def get_connection_string():
        config = settings.SQL_SERVER_CONFIG
        conn_string = (
            f"DRIVER={config['DRIVER']};"
            f"SERVER={config['SERVER']};"
            f"DATABASE={config['DATABASE']};"
            f"UID={config['UID']};"
            f"PWD={config['PWD']};"
            f"TrustServerCertificate={config['TrustServerCertificate']};"
            f"Encrypt={config['Encrypt']};"
        )
        return conn_string
    
    @staticmethod
    def get_database_schema():
        try:
            conn_string = DatabaseService.get_connection_string()
            conn = pyodbc.connect(conn_string)
            cursor = conn.cursor()
            
            query = """
                SELECT 
                    t.TABLE_NAME,
                    c.COLUMN_NAME,
                    c.DATA_TYPE,
                    c.IS_NULLABLE
                FROM 
                    INFORMATION_SCHEMA.TABLES t
                INNER JOIN 
                    INFORMATION_SCHEMA.COLUMNS c ON t.TABLE_NAME = c.TABLE_NAME
                WHERE 
                    t.TABLE_TYPE = 'BASE TABLE'
                ORDER BY 
                    t.TABLE_NAME, c.ORDINAL_POSITION
            """
            
            cursor.execute(query)
            rows = cursor.fetchall()
            
            schema = 'Database Schema:\n\n'
            current_table = ''
            
            for row in rows:
                table_name = row[0]
                column_name = row[1]
                data_type = row[2]
                is_nullable = row[3]
                
                if current_table != table_name:
                    current_table = table_name
                    schema += f'\nTable: {table_name}\nColumns:\n'
                
                nullable = ' (nullable)' if is_nullable == 'YES' else ''
                schema += f'  - {column_name} ({data_type}){nullable}\n'
            
            cursor.close()
            conn.close()
            
            return schema
            
        except Exception as e:
            raise Exception(f"Failed to retrieve database schema: {str(e)}")
    
    @staticmethod
    def execute_query(sql_query):
        try:
            conn_string = DatabaseService.get_connection_string()
            conn = pyodbc.connect(conn_string)
            cursor = conn.cursor()
            
            cursor.execute(sql_query)
            columns = [column[0] for column in cursor.description]
            rows = cursor.fetchall()
            
            results = []
            for row in rows:
                result_dict = {}
                for i, column in enumerate(columns):
                    value = row[i]
                    if value is None:
                        result_dict[column] = None
                    elif isinstance(value, (int, float, str, bool)):
                        result_dict[column] = value
                    else:
                        result_dict[column] = str(value)
                results.append(result_dict)
            
            cursor.close()
            conn.close()
            
            return results
            
        except Exception as e:
            raise Exception(f"Failed to execute query: {str(e)}")
    
    @staticmethod
    def validate_select_query(sql_query):
        sql_upper = sql_query.strip().upper()
        
        if not sql_upper.startswith('SELECT'):
            return False
        
        forbidden_keywords = [
            'INSERT', 'UPDATE', 'DELETE', 'DROP', 'CREATE',
            'ALTER', 'TRUNCATE', 'EXEC', 'EXECUTE', 'MERGE',
            'GRANT', 'REVOKE', 'BACKUP', 'RESTORE'
        ]
        
        for keyword in forbidden_keywords:
            if keyword in sql_upper:
                return False
        
        return True
    
    @staticmethod
    def test_connection():
        try:
            conn_string = DatabaseService.get_connection_string()
            conn = pyodbc.connect(conn_string)
            cursor = conn.cursor()
            cursor.execute("SELECT 1")
            cursor.close()
            conn.close()
            
            return {'success': True, 'message': 'Database connection successful'}
            
        except Exception as e:
            return {'success': False, 'message': f'Database connection failed: {str(e)}'}

from django.conf import settings
from openai import OpenAI
from anthropic import Anthropic

class AIService:
    
    @staticmethod
    def natural_language_to_sql(question, database_schema):
        provider = settings.AI_PROVIDER.lower()
        
        if provider == 'openai':
            return AIService._openai_to_sql(question, database_schema)
        elif provider == 'anthropic':
            return AIService._anthropic_to_sql(question, database_schema)
        else:
            raise Exception(f"Unknown AI provider: {provider}")
    
    @staticmethod
    def _openai_to_sql(question, database_schema):
        try:
            client = OpenAI(api_key=settings.OPENAI_API_KEY)
            
            prompt = f"""You are a SQL expert. Convert the following natural language question to a SQL Server query.

{database_schema}

Question: {question}

Important:
1. Return ONLY the SQL query, no explanations or markdown
2. Use TOP instead of LIMIT for SQL Server
3. Ensure the query is safe and read-only (SELECT only)
4. Use proper SQL Server syntax
5. Do not include code blocks or formatting

SQL Query:"""
            
            response = client.chat.completions.create(
                model="gpt-4",
                messages=[
                    {"role": "system", "content": "You are a SQL expert that converts natural language to SQL Server queries. Return only the SQL query without any explanation."},
                    {"role": "user", "content": prompt}
                ],
                temperature=0,
                max_tokens=500
            )
            
            sql_query = response.choices[0].message.content.strip()
            sql_query = sql_query.replace('```sql', '').replace('```', '').strip()
            
            return sql_query
            
        except Exception as e:
            raise Exception(f"OpenAI API error: {str(e)}")
    
    @staticmethod
    def _anthropic_to_sql(question, database_schema):
        try:
            client = Anthropic(api_key=settings.ANTHROPIC_API_KEY)
            
            prompt = f"""You are a SQL expert. Convert the following natural language question to a SQL Server query.

{database_schema}

Question: {question}

Important:
1. Return ONLY the SQL query, no explanations or markdown
2. Use TOP instead of LIMIT for SQL Server
3. Ensure the query is safe and read-only (SELECT only)
4. Use proper SQL Server syntax
5. Do not include code blocks or formatting

SQL Query:"""
            
            message = client.messages.create(
                model="claude-sonnet-4-20250514",
                max_tokens=1000,
                temperature=0,
                messages=[{"role": "user", "content": prompt}]
            )
            
            sql_query = message.content[0].text.strip()
            sql_query = sql_query.replace('```sql', '').replace('```', '').strip()
            
            return sql_query
            
        except Exception as e:
            raise Exception(f"Anthropic API error: {str(e)}")

from rest_framework.decorators import api_view
from rest_framework.response import Response
from rest_framework import status
from django.conf import settings
from .database_service import DatabaseService
from .ai_service import AIService

@api_view(['GET'])
def health_check(request):
    return Response({
        'status': 'OK',
        'message': 'Django NL2SQL API is running',
        'ai_provider': settings.AI_PROVIDER
    })

@api_view(['GET'])
def test_connection(request):
    result = DatabaseService.test_connection()
    
    if result['success']:
        return Response(result)
    else:
        return Response(result, status=status.HTTP_500_INTERNAL_SERVER_ERROR)

@api_view(['GET'])
def get_schema(request):
    try:
        schema = DatabaseService.get_database_schema()
        return Response({'success': True, 'schema': schema})
    except Exception as e:
        return Response({'success': False, 'error': str(e)}, status=status.HTTP_500_INTERNAL_SERVER_ERROR)

@api_view(['POST'])
def process_query(request):
    try:
        question = request.data.get('question', '').strip()
        
        if not question:
            return Response({'success': False, 'error': 'Question is required'}, status=status.HTTP_400_BAD_REQUEST)
        
        schema = DatabaseService.get_database_schema()
        sql_query = AIService.natural_language_to_sql(question, schema)
        
        if not DatabaseService.validate_select_query(sql_query):
            return Response({
                'success': False,
                'error': 'Only SELECT queries are allowed for security',
                'generatedQuery': sql_query
            }, status=status.HTTP_400_BAD_REQUEST)
        
        results = DatabaseService.execute_query(sql_query)
        
        return Response({
            'success': True,
            'question': question,
            'sql': sql_query,
            'data': results,
            'rowCount': len(results)
        })
        
    except Exception as e:
        return Response({
            'success': False,
            'error': str(e),
            'details': 'Failed to process query'
        }, status=status.HTTP_500_INTERNAL_SERVER_ERROR)

@api_view(['POST'])
def test_sql(request):
    try:
        sql_query = request.data.get('sql', '').strip()
        
        if not sql_query:
            return Response({'success': False, 'error': 'SQL query is required'}, status=status.HTTP_400_BAD_REQUEST)
        
        if not DatabaseService.validate_select_query(sql_query):
            return Response({'success': False, 'error': 'Only SELECT queries are allowed'}, status=status.HTTP_400_BAD_REQUEST)
        
        results = DatabaseService.execute_query(sql_query)
        
        return Response({'success': True, 'data': results, 'rowCount': len(results)})
        
    except Exception as e:
        return Response({'success': False, 'error': str(e)}, status=status.HTTP_500_INTERNAL_SERVER_ERROR)
